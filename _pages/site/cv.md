<script>
  (() => {
    const resume = document.querySelector('[data-resume]');
    if (!resume) return;
    const copies = [...resume.querySelectorAll('[data-copy]')];
    const buttons = [...resume.querySelectorAll('[data-language-button]')];
    const pdfLink = resume.querySelector('[data-pdf-link]');
    const printButton = resume.querySelector('[data-print-button]');
    const config = {
      zh: { pdf: 'https://github.com/James-Yao-xj/MyCV/raw/main/ChineseResume/resume.pdf', print: '打印' },
      en: { pdf: 'https://github.com/James-Yao-xj/MyCV/raw/main/EnglishResume/resume.pdf', print: 'Print' }
    };

    const setLanguage = (language) => {
      resume.dataset.language = language;
      copies.forEach((copy) => { copy.hidden = copy.dataset.copy !== language; });
      buttons.forEach((button) => {
        const active = button.dataset.languageButton === language;
        button.classList.toggle('is-active', active);
        button.setAttribute('aria-pressed', String(active));
      });
      pdfLink.href = config[language].pdf;
      printButton.textContent = config[language].print;
      document.documentElement.lang = language === 'zh' ? 'zh-CN' : 'en';
      try { localStorage.setItem('cv-language', language); } catch (_) {}
    };

    buttons.forEach((button) => button.addEventListener('click', () => setLanguage(button.dataset.languageButton)));
    printButton.addEventListener('click', () => window.print());
    let savedLanguage = 'zh';
    try { savedLanguage = localStorage.getItem('cv-language') || 'zh'; } catch (_) {}
    setLanguage(savedLanguage in config ? savedLanguage : 'zh');
  })();
</script>
